@gun
def deagle():
    """Free Fire Desert Eagle: tall flat-sided silver slide with top rib and flat muzzle face,
    square trigger guard, wide black grip, exposed hammer."""
    # slide with rear serrations and ejection port
    sl = path(('M', -6.2, 3.6), ('L', -6.2, 7.0), ('L', -5.6, 7.6), ('L', 20.6, 7.6), ('L', 21.0, 7.2), ('L', 21.0, 3.6))
    cuts = [cbox((1.6, 4.0, 5.9), (.5, 6.0, 1.6))] + [cbox((1.6, -4.8 + i * .7, 5.4), (.5, .32, 3.2)) for i in range(5)] + \
           [cbox((-1.6, -4.8 + i * .7, 5.4), (.5, .32, 3.2)) for i in range(5)]
    extrude('Slide', sl, 3.2, 'stainless', bevel=.12, cuts=cuts)
    extrude('Barrel', path(('M', 4.0, 7.4), ('L', 20.6, 7.4), ('L', 21.0, 7.0), ('L', 21.0, 3.8), ('L', 4.0, 3.8)), 2.6, 'stainless', bevel=.1)
    extrude('Top rib', rrect(2.0, 7.5, 20.6, 8.1, .2), 1.0, 'stainless', bevel=.05)
    extrude('Front sight', path(('M', 19.0, 8.0), ('L', 20.6, 8.0), ('L', 20.4, 8.9), ('L', 19.5, 8.9)), .5, 'steel', bevel=.03)
    extrude('Rear sight', rrect(-5.4, 7.5, -4.2, 8.5, .15), 2.0, 'steel', bevel=.04)
    lathe('Bore', [(0, .6), (.3, .6)], 'bore', at=(0, 20.9, 5.5), seg=16, bevel=0)
    # frame under the slide, square trigger guard, take-down lever, slide stop
    fr = path(('M', -6.6, 3.7), ('L', 12.8, 3.7), ('L', 12.8, 2.2), ('L', 11.6, 1.8), ('L', 2.8, 1.8), ('L', 1.2, 2.4), ('L', -4.4, 2.4), ('L', -6.6, 3.0))
    extrude('Frame', fr, 3.0, 'stainless', bevel=.1)
    tg = path(('M', 3.0, 2.0), ('L', 3.0, -.8), ('Q', 3.2, -1.6, 4.2, -1.6), ('L', 9.0, -1.6), ('Q', 9.8, -1.5, 10.0, -.8), ('L', 10.4, 2.0), n=4)
    extrude('Trigger guard', band(tg, .7), 1.3, 'stainless', bevel=.08)
    extrude('Trigger', path(('M', 5.0, 2.0), ('L', 5.9, 2.0), ('C', 5.9, .6, 5.6, -.2, 4.8, -.6), ('L', 4.7, -.3), ('C', 5.1, .1, 5.1, 1.0, 5.0, 2.0)), .5, 'steel', bevel=.03)
    lathe('Slide stop', [(0, .55), (.25, .55)], 'steel', at=(-1.55, 1.6, 3.2), axis='X', seg=12, bevel=0)
    extrude('Takedown lever', rrect(6.6, 3.0, 9.6, 3.6, .25), .3, 'steel', at=1.55, bevel=.04)
    # hammer + wide grip with panels
    extrude('Hammer', path(('M', -6.8, 4.4), ('L', -6.0, 4.4), ('L', -6.0, 6.2), ('L', -7.0, 7.4), ('L', -8.0, 7.4), ('L', -7.4, 6.2)), .9, 'steel', bevel=.05)
    gp = path(('M', 1.6, 2.6), ('L', -6.6, 3.2), ('L', -7.0, 1.6), ('C', -7.4, -2.4, -7.6, -5.0, -7.4, -7.0), ('Q', -7.2, -7.8, -6.2, -7.8),
              ('L', -.6, -7.8), ('Q', .4, -7.8, .4, -6.8), ('C', .4, -4.0, .8, -.6, 1.6, 2.6))
    extrude('Grip frame', gp, 3.0, 'stainless', bevel=.12)
    panel = path(('M', .4, 1.6), ('L', -6.2, 2.2), ('C', -6.6, -1.6, -6.8, -4.6, -6.6, -6.8), ('L', -.4, -6.8), ('C', -.3, -3.6, .0, -.6, .4, 1.6))
    extrude('Grip panel', panel, 3.6, 'polymer', crown=.15, inset=.25, bevel=.1)
    lathe('Grip screw', [(0, .32), (.12, .32)], 'steel', at=(1.82, -3.0, -2.0), axis='X', seg=10, bevel=0)
    extrude('Magazine base', rrect(-6.8, -8.6, -.2, -7.7, .3), 3.0, 'steel', bevel=.08)
    return {'bore_z': 5.5, 'muzzle': (21.0, 5.5), 'optic': (8.0, 8.1), 'mag': (-3.4, -4.0)}
