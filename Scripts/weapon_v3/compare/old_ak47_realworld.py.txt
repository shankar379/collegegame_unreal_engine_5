@gun
def ak47():
    BZ = 5.0       # bore axis height
    W = 3.2        # receiver width
    # ---- stamped receiver: flat sides, rear tang, magazine-well dimples, ejection port with the bolt inside
    rec = path(('M', -10.5, 2.4), ('L', -10.5, 7.0), ('L', 18.0, 7.0), ('L', 18.0, 1.6), ('L', 15.6, 1.2),
               ('L', 9.8, 1.2), ('L', 9.8, 1.5), ('L', -6.5, 1.5), ('Q', -9.6, 1.6, -10.5, 2.4))
    cuts = [cbox((W / 2 + .4, 8.6, 6.2), (1.6, 9.2, 1.05)),            # ejection port (right)
            cbox((W / 2 + .2, 12.8, 3.0), (.32, 4.4, 2.2)),              # magwell dimple R
            cbox((-W / 2 - .2, 12.8, 3.0), (.32, 4.4, 2.2))]             # magwell dimple L
    extrude('Receiver', rec, W, 'steel', bevel=.12, cuts=cuts)
    extrude('Bolt carrier', rrect(3.6, 5.55, 13.4, 6.85, .3), 1.9, 'steel_bright', at=.25, skin=False, bevel=.06)
    lathe('Bolt face', [(0, .42), (.5, .42), (.6, .3)], 'steel_bright', at=(.9, 13.2, 6.05), axis='Y', seg=12)
    # trunnion block at the front of the receiver
    extrude('Front trunnion', rrect(15.5, 1.8, 19.2, 7.6, .4), W + .3, 'steel', bevel=.1)

    # ---- dust cover: rounded top sheet with stamped ribs and the rear release button
    dc = arc(0, 6.85, 1.66, 180, 0, 16)
    extrude('Dust cover', dc, 26.5, 'steel', plane='XZ', at=4.6, bevel=.08, skin=True)
    for y in (-3.5, 2.5, 8.0):
        ribs = [(x * .92, 6.85 + (z - 6.85) * 1.12) for x, z in arc(0, 6.85, 1.62, 165, 15, 10)]
        extrude('Cover rib', band(ribs, .32), .9, 'steel', plane='XZ', at=y, bevel=.05)
    lathe('Cover button', [(0, .55), (.5, .55), (.7, .4), (.75, .2)], 'steel', at=(0, -8.65, 7.45), axis='Y', seg=12)

    # ---- charging handle (right) and the long selector lever (right)
    lathe('Charging handle', [(0, .32), (1.0, .32), (1.15, .48), (1.9, .55), (2.1, .3)], 'steel_bright', at=(1.0, 12.4, 6.35), axis='X', seg=12)
    sel = path(('M', -6.6, 4.3), ('L', -5.4, 4.3), ('L', -4.6, 5.75), ('L', 2.6, 6.2), ('L', 2.8, 6.85), ('L', -4.9, 6.75), ('L', -6.2, 6.3), ('L', -7.0, 5.0))
    extrude('Selector lever', sel, .22, 'steel', at=W / 2 + .14, bevel=.04)
    lathe('Selector pivot', [(0, .5), (.2, .5), (.3, .28)], 'steel', at=(W / 2 + .1, -5.6, 5.6), axis='X', seg=14, bevel=0)

    # ---- rivets and pins (both sides)
    for y, z in ((-8.2, 2.6), (-8.2, 4.8), (-1.2, 2.3), (2.8, 2.3), (6.3, 2.3), (16.2, 2.6), (17.3, 4.9), (15.2, 5.6)):
        rivet(W / 2, y, z, side=1)
        rivet(-W / 2, y, z, side=-1)

    # ---- pistol grip (bakelite-brown wood), rounded, thinner at the bottom
    gp = path(('M', 2.6, 1.5), ('L', -1.6, 1.5), ('C', -3.0, -2.0, -4.6, -6.0, -5.6, -9.2), ('Q', -5.9, -10.3, -4.8, -10.4),
              ('L', -2.2, -10.0), ('Q', -1.2, -9.7, -1.0, -8.7), ('C', -.2, -5.0, 1.4, -1.6, 2.6, 1.5))
    extrude('Grip', gp, 2.9, 'wood_dark', crown=.35, inset=.45, taper=lambda a, b: 1 - max(0, -b) * .012, bevel=.1)

    # ---- trigger guard (stamped strip) and trigger
    tg = path(('M', .8, 1.7), ('L', 1.0, -1.6), ('Q', 1.4, -3.1, 3.2, -3.1), ('L', 8.4, -3.1), ('Q', 9.6, -3.0, 9.7, -1.4), ('L', 9.7, 1.7), n=8)
    extrude('Trigger guard', band(tg, .42), 1.05, 'steel', bevel=.06)
    trg = path(('M', 3.6, 1.5), ('L', 4.4, 1.5), ('C', 4.4, -.4, 4.0, -1.6, 3.0, -2.2), ('L', 2.8, -1.9), ('C', 3.5, -1.0, 3.7, .2, 3.6, 1.5))
    extrude('Trigger', trg, .5, 'steel_bright', bevel=.04)
    extrude('Mag catch', path(('M', 8.5, 1.4), ('L', 9.7, 1.4), ('L', 9.6, -1.6), ('L', 9.0, -2.1), ('L', 8.6, -1.6)), 1.5, 'steel', bevel=.06)

    # ---- curved 30-round magazine with stamped ribs and floorplate
    mg = path(('M', 10.0, 1.4), ('L', 16.0, 1.4), ('C', 16.3, -5.0, 19.6, -11.6, 25.6, -17.6), ('L', 20.4, -21.0),
              ('C', 15.6, -15.0, 10.8, -8.0, 10.0, 1.4), n=14)
    extrude('Magazine', mg, 2.55, 'steel', taper=lambda a, b: 1 - max(0, -b) * .004, bevel=.12)
    for t in (.28, .62):
        line = []
        for i in range(13):
            s = i / 12
            fy = 16.0 * (1 - s) ** 3 + 3 * 16.3 * (1 - s) ** 2 * s + 3 * 19.6 * (1 - s) * s * s + 25.6 * s ** 3
            fz = 1.4 * (1 - s) ** 3 + 3 * -5.0 * (1 - s) ** 2 * s + 3 * -11.6 * (1 - s) * s * s + -17.6 * s ** 3
            by = 10.0 * (1 - s) ** 3 + 3 * 10.8 * (1 - s) ** 2 * s + 3 * 15.6 * (1 - s) * s * s + 20.4 * s ** 3
            bz = 1.4 * (1 - s) ** 3 + 3 * -8.0 * (1 - s) ** 2 * s + 3 * -15.0 * (1 - s) * s * s + -21.0 * s ** 3
            line.append((by + (fy - by) * t, bz + (fz - bz) * t))
        line = line[1:-1]
        for side in (1, -1):
            extrude('Mag rib', band(line, .55), .3, 'steel', at=side * 1.3, bevel=.08, skin=True)
    extrude('Mag floorplate', band([(19.9, -21.25), (26.1, -17.4)], .7), 2.85, 'steel', bevel=.1, skin=True)
    extrude('Mag lug', rrect(15.6, .2, 16.6, 1.6, .2), 1.6, 'steel', bevel=.05)

    # ---- wooden buttstock with steel butt plate
    st = path(('M', -10.3, 6.9), ('L', -36.0, 5.4), ('L', -36.0, -4.2), ('C', -30, -2.6, -22, -.4, -14.0, 1.2), ('Q', -11.0, 1.7, -10.3, 2.6), n=10)
    extrude('Stock', st, 3.6, 'wood', crown=.35, inset=.55, taper=lambda a, b: 1 + max(0, -a - 10) * .004, bevel=.12)
    bp = path(('M', -36.0, 5.5), ('L', -36.9, 5.5), ('Q', -37.3, .6, -36.9, -4.4), ('L', -36.0, -4.3))
    extrude('Butt plate', bp, 3.95, 'steel', bevel=.1)
    extrude('Stock tang', rrect(-13.0, 5.6, -10.0, 6.95, .2), 3.7, 'steel', bevel=.05)

    # ---- handguards: lower (wood, palm swells), upper over the gas tube
    lh = path(('M', 19.2, 6.4), ('L', 33.6, 6.4), ('L', 33.6, 3.0), ('Q', 33.4, 2.0, 32.2, 2.0), ('L', 21.0, 2.0), ('Q', 19.4, 2.1, 19.2, 3.4))
    extrude('Lower handguard', lh, 4.3, 'wood', crown=.45, inset=.6, taper=lambda a, b: .9 + .1 * math.sin(max(0, min(1, (a - 19.2) / 14.4)) * math.pi), bevel=.12)
    for y in (19.0, 33.9):
        extrude('Handguard retainer', rrect(y - .45, 1.7, y + .45, 6.7, .25), 4.5 if y < 30 else 3.6, 'steel', bevel=.08)
    lathe('Gas tube', [(0, .95), (12.6, .95)], 'steel', at=(0, 20.6, 7.6), seg=18, bevel=.03)
    uh = [(x, z) for x, z in arc(0, 7.25, 1.7, 180, 0, 14)]
    extrude('Upper handguard', uh, 9.4, 'wood', plane='XZ', at=25.0, bevel=.15)
    for y in (29.95, 30.55):
        lathe('Gas port ring', [(0, 1.02), (.3, 1.02)], 'steel', at=(0, y, 7.6), seg=18, bevel=0)

    # ---- rear sight (leaf on a block over the trunnion)
    extrude('Rear sight block', path(('M', 17.8, 7.0), ('L', 22.4, 7.0), ('L', 22.4, 8.6), ('L', 18.6, 8.6), ('L', 17.8, 7.9)), 2.4, 'steel', bevel=.08)
    extrude('Rear sight leaf', path(('M', 18.4, 8.6), ('L', 23.0, 9.05), ('L', 23.0, 9.55), ('L', 18.4, 9.2)), 1.5, 'steel', bevel=.04)
    extrude('Sight slider', rrect(20.0, 8.65, 21.0, 9.75, .15), 1.9, 'steel', bevel=.04)

    # ---- barrel with gas block, front sight, cleaning rod and slant muzzle brake
    lathe('Barrel', [(0, 1.25), (1.5, 1.25), (1.8, 1.0), (24.0, .95), (24.0, .88), (33.2, .82)], 'blued', at=(0, 19.2, BZ), seg=20, bevel=.03)
    gb = path(('M', 33.0, 3.9), ('L', 33.0, 8.7), ('L', 34.7, 8.7), ('L', 37.0, 6.1), ('L', 37.0, 3.9))
    extrude('Gas block', gb, 2.3, 'steel', bevel=.1)
    fsb = path(('M', 47.0, 3.4), ('L', 47.0, 6.4), ('L', 48.0, 7.4), ('L', 50.0, 7.4), ('L', 50.6, 6.2), ('L', 50.6, 3.4))
    extrude('Front sight base', fsb, 2.2, 'steel', bevel=.08)
    for x in (-.85, .85):
        extrude('Sight ear', path(('M', 47.9, 7.0), ('L', 49.9, 7.0), ('L', 49.6, 9.9), ('L', 48.4, 9.9)), .34, 'steel', at=x, bevel=.03)
    lathe('Front post', [(0, .16), (2.4, .12)], 'steel', at=(0, 49.0, 7.2), axis='Z', seg=8, bevel=0)
    extrude('Bayonet lug', rrect(47.4, 2.2, 50.0, 3.6, .3), 1.0, 'steel', bevel=.05)
    lathe('Cleaning rod', [(0, .26), (30.0, .26), (30.2, .2)], 'steel_bright', at=(0, 19.8, 3.2), seg=10, bevel=0)
    lathe('Rod head', [(0, .4), (.8, .4)], 'steel_bright', at=(0, 50.0, 3.2), seg=10, bevel=0)
    slant = cprism([(53.6, BZ + 1.5), (54.9, BZ + 1.5), (54.9, BZ + .2), (54.0, BZ - .2)], 4, 'YZ', 0)
    lathe('Muzzle brake', [(0, .62), (0, 1.12), (3.9, 1.12), (3.9, .62)], 'blued', at=(0, 51.0, BZ), seg=24, closed=True, bevel=.05, cuts=[slant])
    return {'bore_z': BZ, 'muzzle': (54.9, BZ), 'optic': (8.0, 8.6), 'mag': (13.5, -8.0)}
